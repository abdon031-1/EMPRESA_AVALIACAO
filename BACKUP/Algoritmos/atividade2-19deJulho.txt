<script>

// Atividade 1 - Média Aritmética

function calcularMA(nota1, nota2, nota3, nota4) {
    return (nota1 + nota2 + nota3 + nota4) / 4;
}

let nota1 = Number(prompt("Digite a primeira nota: "));
let nota2 = Number(prompt("Digite a segunda nota: "));
let nota3 = Number(prompt("Digite a terceira nota: "));
let nota4 = Number(prompt("Digite a quarta nota: "));

let resultadoMA = calcularMA(nota1, nota2, nota3, nota4);

alert("A média aritmética é: " + resultadoMA);


// Atividade 2 - Média Ponderada

function calcularMP(nota1, peso1, nota2, peso2) {
    return (nota1 * peso1 + nota2 * peso2) / (peso1 + peso2);
}

let nota5 = Number(prompt("Digite a primeira nota: "));
let peso1 = Number(prompt("Digite o peso da primeira nota: "));

let nota6 = Number(prompt("Digite a segunda nota: "));
let peso2 = Number(prompt("Digite o peso da segunda nota: "));

let resultadoMP = calcularMP(nota5, peso1, nota6, peso2);

alert("A média ponderada é: " + resultadoMP);


// Atividade 3 - Áreas

function quadrado(lado) {
    return lado * lado;
}

function triangulo(base, altura) {
    return (base * altura) / 2;
}

function circulo(raio) {
    return Math.PI * raio * raio;
}

let lado = Number(prompt("Digite o lado do quadrado: "));
let base = Number(prompt("Digite a base do triângulo: "));
let altura = Number(prompt("Digite a altura do triângulo: "));
let raio = Number(prompt("Digite o raio do círculo: "));

let areaQuadrado = quadrado(lado);
let areaTriangulo = triangulo(base, altura);
let areaCirculo = circulo(raio);

alert(
    "Área do quadrado: " + areaQuadrado +
    "\nÁrea do triângulo: " + areaTriangulo +
    "\nÁrea do círculo: " + areaCirculo
);

</script>
